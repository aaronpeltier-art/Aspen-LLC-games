  <div id="hud">

  <canvas id="gameCanvas" width="600" height="400"></canvas>

  <div class="logs">
    [SYSTEM]: WASD to Move | SHIFT to Sprint | Return to Blue Zone to Sell
  </div>

  <script>
    const canvas = document.getElementById("gameCanvas");
    const ctx = canvas.getContext("2d");

    // Game State
    let state = {
      score: 0,
      heldScrap: 0,
      stamina: 100,
      battery: 100,
      gameOver: false,
      time: 0
    };

    // Entities
    const player = { x: 520, y: 320, size: 14, speed: 2.5, isSprinting: false };
    const ship = { x: 480, y: 280, w: 120, h: 120 }; // Safe Zone
    const monster = { x: 40, y: 40, speed: 1.2, anger: 1 };
    let scraps = [];

    // Initialize Scrap Types
    const scrapTypes = [
      { name: 'Bolt', val: 15, color: '#888', size: 4 },
      { name: 'V-Jar', val: 45, color: '#00ff00', size: 6 },
      { name: 'Engine', val: 110, color: '#ff4400', size: 8 }
    ];

    function spawnScrap() {
      let t = scrapTypes[Math.floor(Math.random() * scrapTypes.length)];
      scraps.push({
        x: Math.random() * 400 + 40,
        y: Math.random() * 240 + 40,
        ...t
      });
    }

    // Start with some scrap on the map
    for(let i=0; i<6; i++) spawnScrap();

    // Input Handling
    const keys = {};
    window.onkeydown = (e) => keys[e.key.toLowerCase()] = true;
    window.onkeyup = (e) => keys[e.key.toLowerCase()] = false;

    function update() {
      if (state.gameOver) return;
      state.time += 0.01;

      // 1. Movement & Stamina Logic
      let isMoving = keys['w'] || keys['a'] || keys['s'] || keys['d'];
      player.isSprinting = keys['shift'] && state.stamina > 0 && isMoving;
      
      let currentSpeed = player.isSprinting ? player.speed * 2 : player.speed;
      
      if (player.isSprinting) {
        state.stamina -= 0.6;
      } else if (state.stamina < 100) {
        state.stamina += 0.25;
      }

      if (keys['w'] && player.y > 0) player.y -= currentSpeed;
      if (keys['s'] && player.y < canvas.height - player.size) player.y += currentSpeed;
      if (keys['a'] && player.x > 0) player.x -= currentSpeed;
      if (keys['d'] && player.x < canvas.width - player.size) player.x += currentSpeed;

      // 2. Flashlight Battery
      if (state.battery > 0) state.battery -= 0.03;

      // 3. Monster AI (Lethal stalking)
      let dx = (player.x + player.size/2) - monster.x;
      let dy = (player.y + player.size/2) - monster.y;
      let dist = Math.sqrt(dx*dx + dy*dy);
      
      // Monster gets faster as the game goes on and based on loot weight
      let difficulty = 1 + (state.heldScrap / 300) + (state.time / 20);
      monster.x += (dx/dist) * monster.speed * difficulty;
      monster.y += (dy/dist) * monster.speed * difficulty;

      // 4. Picking up Scrap
      scraps.forEach((s, i) => {
        let sDist = Math.hypot(player.x - s.x, player.y - s.y);
        if (sDist < 20) {
          state.heldScrap += s.val;
          scraps.splice(i, 1);
          spawnScrap(); // Maintain scrap density
        }
      });

      // 5. The Ship (Secure Quota)
      if (player.x > ship.x && player.y > ship.y) {
        state.score += state.heldScrap;
        state.heldScrap = 0;
      }

      // 6. Death Condition
      if (dist < 15) {
        state.gameOver = true;
        alert("TERMINATED. YOU FAILED THE COMPANY.\nFinal Quota Met: $" + state.score);
        location.reload();
      }

      // 7. Update HUD
      document.getElementById("score").innerText = state.score + " (+$" + state.heldScrap + ")";
      document.getElementById("stamina-bar").style.width = state.stamina + "%";
      document.getElementById("battery-bar").style.width = state.battery + "%";
      document.getElementById("warning").innerText = dist < 120 ? "NEARBY" : "";
    }

    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // Draw Industry Ship (Blue Zone)
      ctx.fillStyle = "#112244";
      ctx.fillRect(ship.x, ship.y, ship.w, ship.h);
      ctx.strokeStyle = "#00ccff";
      ctx.strokeRect(ship.x, ship.y, ship.w, ship.h);
      ctx.fillStyle = "#00ccff";
      ctx.font = "10px monospace";
      ctx.fillText("SAFETY / SHIP", ship.x + 15, ship.y + 20);

      // Draw Scrap Items
      scraps.forEach(s => {
        ctx.fillStyle = s.color;
        ctx.shadowBlur = 5;
        ctx.shadowColor = s.color;
        ctx.beginPath();
        ctx.rect(s.x, s.y, s.size, s.size);
        ctx.fill();
        ctx.shadowBlur = 0;
      });

      // Draw Player (The Animal)
      ctx.fillStyle = "#ffaa00";
      ctx.fillRect(player.x, player.y, player.size, player.size);

      // Draw Monster (The Beast)
      ctx.fillStyle = "red";
      ctx.font = "24px Arial";
      ctx.fillText("👹", monster.x - 12, monster.y + 8);

      // FLASHLIGHT SYSTEM
      // The radius shrinks as battery dies
      let viewDist = state.battery > 0 ? 160 : 40; 
      let grad = ctx.createRadialGradient(
        player.x + 7, player.y + 7, 0, 
        player.x + 7, player.y + 7, viewDist
      );
      grad.addColorStop(0, 'rgba(0,0,0,0)');
      grad.addColorStop(0.7, 'rgba(0,0,0,0.2)');
      grad.addColorStop(1, 'rgba(0,0,0,0.98)');
      
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      update();
      requestAnimationFrame(draw);
    }

    draw();
  </script>
